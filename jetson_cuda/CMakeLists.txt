cmake_minimum_required(VERSION 3.10)
project(pirate_xavier LANGUAGES CXX CUDA)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
set(PIRATE_CUDA_ARCH "72" CACHE STRING "CUDA SM number; Xavier=72, RTX3080=86")
add_executable(pirate_bench pirate_bench.cu)
set_target_properties(pirate_bench PROPERTIES CUDA_STANDARD 14 CUDA_STANDARD_REQUIRED YES CXX_STANDARD 14 CUDA_ARCHITECTURES OFF)
target_compile_options(pirate_bench PRIVATE "$<$<COMPILE_LANGUAGE:CUDA>:-O3;--fmad=false;-gencode=arch=compute_${PIRATE_CUDA_ARCH},code=sm_${PIRATE_CUDA_ARCH}>")
enable_testing()
add_test(NAME gpu_self_test COMMAND pirate_bench --self-test)
